# netgen 6.2.2101 static libraries for pySMESH: ngcore, netgen, nglib, and zlib.
#
# netgen's own CMake is not used. It hard-codes SHARED libraries for ngcore and nglib, names
# the OCCT toolkits of before 7.8, needs git for its version header and turns its superbuild
# on. These targets compile the same sources as looooo/SMESH's cmake/Netgen/CMakeLists.txt
# (third_party_ref/looooo_SMESH), as STATIC libraries linked into _core.pyd.
#
# ${NETGEN_SRC_DIR} is the staged, patched tree (prepare.py: staged/src/Netgen). zlib is
# compiled from extern/zlib, which no patch touches.
#
# Defines. The root CMakeLists defines its global set for every target (WIN32, NOMINMAX,
# _USE_MATH_DEFINES, ...). PYSMESH_NETGEN_DEFINITIONS adds what netgen needs, and every
# translation unit that includes a netgen header gets exactly this list: these libraries,
# NETGENPlugin, the binding that includes the plugin headers and the probe. A difference
# between them changes the layout of netgen classes silently: NO_PARALLEL_THREADS, for
# example, replaces the std::mutex inside netgen::Mesh by an empty class (parthreads.hpp).
# It is not defined anywhere, as in netgen's own Windows build. MSVC_EXPRESS keeps
# mystdlib.h from including the MFC headers. The three *_EXPORTS put DLL_HEADER on its
# dllexport branch, as the root CMakeLists does for SMESH, so that data such as mparam is
# not imported from a DLL that does not exist. NETGEN_V6 and NETGENPLUGIN_EXPORTS are the
# plugin's (its V6 code path, and NETGENPLUGIN_EXPORT on its dllexport branch); netgen
# does not read them, and they are in the one list so that the sets cannot differ.
set(NETGEN_SRC_DIR ${STAGED_DIR}/src/Netgen)
set(ZLIB_SRC_DIR ${CMAKE_SOURCE_DIR}/extern/zlib)

set(PYSMESH_NETGEN_DEFINITIONS
    OCCGEOMETRY MSVC_EXPRESS HAVE_STRUCT_TIMESPEC _WIN32_WINNT=0x0A00
    NGCORE_EXPORTS NGLIB_EXPORTS NGINTERFACE_EXPORTS
    NETGEN_V6 NETGENPLUGIN_EXPORTS)

# netgen_version.hpp is generated by netgen's CMake from git describe; core/version.cpp and
# meshing/global.cpp include it.
set(PYSMESH_NETGEN_VERSION 6.2.2101)
set(PYSMESH_NETGEN_VERSION_HASH g5e489319)
set(NETGEN_GENERATED_DIR ${CMAKE_BINARY_DIR}/netgen_generated)
configure_file(${CMAKE_SOURCE_DIR}/cmake/Netgen/netgen_version.hpp.in
               ${NETGEN_GENERATED_DIR}/netgen_version.hpp @ONLY)

set(PYSMESH_NETGEN_INCLUDE_DIRS
    ${NETGEN_SRC_DIR}/libsrc
    ${NETGEN_SRC_DIR}/libsrc/include
    ${NETGEN_SRC_DIR}/libsrc/general
    ${NETGEN_SRC_DIR}/nglib
    ${NETGEN_GENERATED_DIR})

# MSVC warnings in the netgen sources that the code base relies on: C4005 (mystdlib.h
# redefines WIN32_LEAN_AND_MEAN, which the global set defines) and C4068 (unknown pragmas
# of other compilers).
set(PYSMESH_NETGEN_OPTIONS /wd4005 /wd4068)

foreach(_ng_target IN ITEMS ngcore netgen nglib)
    target_compile_definitions(${_ng_target} PRIVATE ${PYSMESH_NETGEN_DEFINITIONS})
    target_compile_options(${_ng_target} PRIVATE ${PYSMESH_NETGEN_OPTIONS})
    target_include_directories(${_ng_target} PRIVATE ${PYSMESH_NETGEN_INCLUDE_DIRS})
endforeach()

# The OCCT toolkits of looooo's list, renamed for OCCT 8 (TKSTEP -> TKDESTEP, TKIGES ->
# TKDEIGES) and without the toolkits OCCT 8 merged or dropped. TKDESTL is left out on
# purpose: patches/netgen/occgeom_save_without_stl.patch removes netgen's only use of it,
# so that _core.pyd does not import TKDESTL.dll, which the wheel does not ship.
target_link_libraries(ngcore PUBLIC Threads::Threads)
target_link_libraries(netgen PUBLIC
    ngcore pysmesh_zlib
    TKernel TKMath TKG2d TKG3d TKGeomBase TKBRep TKGeomAlgo TKTopAlgo TKPrim TKBO TKBool
    TKShHealing TKOffset TKMesh TKXSBase TKDESTEP TKDEIGES TKLCAF TKCAF TKXCAF)
target_link_libraries(nglib PUBLIC netgen)

set(NETGEN_STATIC_LIBS nglib netgen ngcore pysmesh_zlib)
